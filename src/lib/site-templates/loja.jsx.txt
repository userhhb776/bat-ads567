function useCatalog() {
  const [data, setData] = useState(DEFAULT);
  const [loaded, setLoaded] = useState(false);
  const [source, setSource] = useState("modelo");
  const load = async () => {
    if (supabase && validKey) {
      const r = await supabase.from("site_stores").select("data").eq("site_key", STORE_KEY).maybeSingle();
      if (r.data && r.data.data) { setData(merge(r.data.data)); setSource("banco"); setLoaded(true); return; }
    }
    try { const raw = localStorage.getItem(LOCAL); if (raw) { setData(merge(JSON.parse(raw))); setSource("aparelho"); } } catch (e) {}
    setLoaded(true);
  };
  useEffect(() => { load(); }, []);
  return { data, setData, loaded, source, reload: load };
}

}

    if (online) {
      if (!session) { setSaving(false); flash("Entre para salvar."); return; }
      const r = await supabase.from("site_stores").upsert({ user_id: session.user.id, site_key: STORE_KEY, data: data }, { onConflict: "site_key" });
      setSaving(false);
      if (r.error) { flash("Erro ao salvar: " + r.error.message); return; }
    } else { localStorage.setItem(LOCAL, JSON.stringify(data)); setSaving(false); }
    cat.setData(merge(data)); setDraft(data); flash("Loja atualizada!");
  };
  const set = (k, v) => setDraft(Object.assign({}, draft, { [k]: v }));
  const inp = "mt-1 h-11 w-full rounded-xl border bg-white px-3 text-sm";

  if (online && !session) return (
    <div className="flex min-h-screen items-center justify-center bg-gray-50 p-5" style={{ fontFamily: "'Plus Jakarta Sans', sans-serif" }}>
      <form onSubmit={async (e) => { e.preventDefault(); const r = await supabase.auth.signInWithPassword(login); if (r.error) flash("E-mail ou senha incorretos."); }} className="w-full max-w-sm space-y-4 rounded-3xl bg-white p-7 shadow-xl">
        <div className="text-center"><span className="mx-auto mb-3 flex h-12 w-12 items-center justify-center rounded-2xl bg-gray-900 text-white"><Lock className="h-5 w-5" /></span><h1 className="text-2xl font-extrabold">Admin · {cat.data.brand}</h1><p className="text-sm text-gray-500">Entre com a conta do painel</p></div>
        <input type="email" required placeholder="E-mail" value={login.email} onChange={(e) => setLogin(Object.assign({}, login, { email: e.target.value }))} className={inp} />
        <input type="password" required placeholder="Senha" value={login.password} onChange={(e) => setLogin(Object.assign({}, login, { password: e.target.value }))} className={inp} />
        <button className="h-12 w-full rounded-xl bg-gray-900 font-bold text-white">Entrar</button>
        {msg && <p className="text-center text-sm text-red-600">{msg}</p>}
      </form>
    </div>
  );

  const tabs = [["produtos", "Produtos", Package], ["pedidos", "Pedidos", Receipt], ["banners", "Banners", GalleryHorizontal], ["categorias", "Categorias", LayoutGrid], ["cupons", "Cupons", TicketPercent], ["loja", "Loja", Settings]];
  const newProduct = { id: uid(), name: "", price: 0, oldPrice: 0, category: (draft.categories[0] || {}).name || "", image: "", rating: 5, stock: 10, active: true, description: "" };
  const Row = ({ children }) => <div className="flex items-center gap-3 rounded-2xl border bg-white p-3">{children}</div>;
  const listEditor = (key, blank, fields) => (
    <div className="space-y-3">{draft[key].map((it, i) => <div key={i} className="space-y-2 rounded-2xl border bg-white p-4">
      <div className="grid gap-2 sm:grid-cols-2">{fields.map(([f, l, type]) => <label key={f} className={"block text-xs font-semibold text-gray-500 " + (f === "image" ? "sm:col-span-2" : "")}>{l}{type === "bool" ? <input type="checkbox" className="ml-2 align-middle" checked={it[f] !== false} onChange={(e) => set(key, draft[key].map((x, j) => j === i ? Object.assign({}, x, { [f]: e.target.checked }) : x))} /> : <input type={type || "text"} value={it[f] == null ? "" : it[f]} onChange={(e) => set(key, draft[key].map((x, j) => j === i ? Object.assign({}, x, { [f]: type === "number" ? Number(e.target.value) : e.target.value }) : x))} className={inp} />}</label>)}</div>
      {it.image && <img src={it.image} alt="" className="h-20 w-32 rounded-xl object-cover" />}
      <button onClick={() => set(key, draft[key].filter((_, j) => j !== i))} className="text-xs font-bold text-red-600">Remover</button></div>)}
      <button onClick={() => set(key, draft[key].concat([blank]))} className="flex h-11 w-full items-center justify-center gap-2 rounded-2xl border-2 border-dashed text-sm font-bold"><Plus className="h-4 w-4" /> Adicionar</button>
      <button disabled={saving} onClick={() => save()} className="h-12 w-full rounded-xl bg-gray-900 font-bold text-white">{saving ? "Salvando…" : "Salvar alterações"}</button>
    </div>
  );

  return (
    <div className="min-h-screen bg-gray-50 pb-24 text-gray-900" style={{ fontFamily: "'Plus Jakarta Sans', sans-serif" }}>
      <header className="sticky top-0 z-30 border-b bg-white"><div className="mx-auto flex h-14 max-w-5xl items-center gap-3 px-4"><span className="font-extrabold">Admin · {draft.brand}</span><span className={"rounded-full px-2 py-0.5 text-[11px] font-bold " + (online ? "bg-green-100 text-green-700" : "bg-amber-100 text-amber-700")}>{online ? "Salvando no banco" : "Modo local"}</span>{session && <button onClick={() => supabase.auth.signOut()} className="ml-auto text-sm font-semibold text-gray-500">Sair</button>}</div>
        <nav className="mx-auto flex max-w-5xl gap-1 overflow-x-auto px-3 pb-2">{tabs.map(([k, l, Ic]) => <button key={k} onClick={() => setTab(k)} className={"flex shrink-0 items-center gap-1.5 rounded-full px-3 py-2 text-sm font-bold " + (tab === k ? "bg-gray-900 text-white" : "text-gray-600")}><Ic className="h-4 w-4" />{l}</button>)}</nav></header>
      {!online && <p className="mx-auto mt-4 max-w-5xl px-4 text-sm text-amber-700">Modo local: as mudanças ficam só neste aparelho. Publique pelo Bat Dev V-0 para salvar no banco e aparecer para os clientes.</p>}
      <main className="mx-auto max-w-5xl space-y-4 p-4">
        {tab === "produtos" && <>
          <div className="grid grid-cols-3 gap-2">{[["Produtos", draft.products.length], ["Ativos", draft.products.filter((p) => p.active !== false).length], ["Sem estoque", draft.products.filter((p) => p.stock <= 0).length]].map(([l, v]) => <div key={l} className="rounded-2xl border bg-white p-3 text-center"><p className="text-2xl font-extrabold">{v}</p><p className="text-xs text-gray-500">{l}</p></div>)}</div>
          <button onClick={() => setEdit(newProduct)} className="flex h-12 w-full items-center justify-center gap-2 rounded-xl bg-gray-900 font-bold text-white"><Plus className="h-4 w-4" /> Novo produto</button>
          {draft.products.map((p) => <Row key={p.id}><img src={p.image} alt="" className="h-14 w-14 rounded-xl object-cover" /><div className="min-w-0 flex-1"><p className="truncate font-bold">{p.name}</p><p className="text-sm text-gray-500">{brl(p.price)} · {p.stock} un · {p.category}{p.active === false ? " · oculto" : ""}</p></div><button onClick={() => setEdit(p)} aria-label="Editar" className="flex h-10 w-10 items-center justify-center rounded-full border"><Pencil className="h-4 w-4" /></button></Row>)}
        </>}
        {tab === "pedidos" && (!online ? <p className="text-sm text-gray-500">Os pedidos aparecem aqui quando a loja estiver publicada e salvando no banco.</p> : <>
          <button onClick={loadOrders} className="flex items-center gap-2 text-sm font-bold"><RefreshCw className="h-4 w-4" /> Atualizar</button>
          {orders.length === 0 && <p className="text-sm text-gray-500">Nenhum pedido ainda.</p>}
          {orders.map((o) => <div key={o.id} className="space-y-2 rounded-2xl border bg-white p-4"><div className="flex items-center justify-between gap-2"><p className="font-bold">{(o.customer || {}).name || "Cliente"}</p><p className="font-extrabold">{brl(o.total)}</p></div>
            <p className="text-xs text-gray-500">{new Date(o.created_at).toLocaleString("pt-BR")} · {(o.customer || {}).phone}</p>
            <ul className="text-sm">{(o.items || []).map((x, i) => <li key={i}>{x.qty}x {x.name}</li>)}</ul><p className="text-xs text-gray-500">{(o.customer || {}).address}</p>
            <select value={o.status} onChange={async (e) => { const st = e.target.value; const r = await supabase.from("site_orders").update({ status: st }).eq("id", o.id); if (r.error) flash(r.error.message); else setOrders(orders.map((x) => x.id === o.id ? Object.assign({}, x, { status: st }) : x)); }} className="h-10 rounded-xl border px-3 text-sm">{["novo", "pago", "enviado", "entregue", "cancelado"].map((s) => <option key={s} value={s}>{s}</option>)}</select></div>)}
        </>)}
        {tab === "banners" && listEditor("banners", { title: "", text: "", image: "", cta: "Comprar" }, [["title", "Título"], ["text", "Texto"], ["cta", "Botão"], ["image", "Link da imagem"]])}
        {tab === "categorias" && listEditor("categories", { name: "", image: "" }, [["name", "Nome"], ["image", "Link da imagem"]])}
        {tab === "cupons" && listEditor("coupons", { code: "", percent: 10, active: true }, [["code", "Código"], ["percent", "Desconto (%)", "number"], ["active", "Ativo", "bool"]])}
        {tab === "loja" && <div className="space-y-3 rounded-2xl border bg-white p-4">
          {[["brand", "Nome da loja"], ["announce", "Faixa do topo"], ["whatsapp", "WhatsApp dos pedidos (com DDD)"], ["instagram", "Instagram"], ["email", "E-mail"], ["address", "Cidade / endereço"]].map(([k, l]) => <label key={k} className="block text-xs font-semibold text-gray-500">{l}<input value={draft[k] || ""} onChange={(e) => set(k, e.target.value)} className={inp} /></label>)}
          <div className="grid grid-cols-2 gap-3"><label className="block text-xs font-semibold text-gray-500">Parcelas<input type="number" value={draft.installments} onChange={(e) => set("installments", Number(e.target.value) || 1)} className={inp} /></label><label className="block text-xs font-semibold text-gray-500">Frete grátis a partir de (R$)<input type="number" value={draft.freeShippingFrom} onChange={(e) => set("freeShippingFrom", Number(e.target.value) || 0)} className={inp} /></label></div>
          <div className="grid grid-cols-2 gap-3">{[["accent", "Cor principal"], ["ink", "Cor do texto"]].map(([k, l]) => <label key={k} className="flex items-center gap-2 text-xs font-semibold text-gray-500"><input type="color" value={draft.colors[k]} onChange={(e) => set("colors", Object.assign({}, draft.colors, { [k]: e.target.value }))} className="h-10 w-12 rounded" />{l}</label>)}</div>
          <button disabled={saving} onClick={() => save()} className="h-12 w-full rounded-xl bg-gray-900 font-bold text-white">{saving ? "Salvando…" : "Salvar alterações"}</button>
        </div>}
      </main>
      {edit && <div className="fixed inset-0 z-50 flex items-end justify-center sm:items-center" onClick={() => setEdit(null)}><div className="absolute inset-0 bg-black/50" /><div onClick={(e) => e.stopPropagation()} className="relative max-h-[92dvh] w-full max-w-lg space-y-3 overflow-y-auto rounded-t-3xl bg-white p-5 sm:rounded-3xl">
        <div className="flex items-center justify-between"><h3 className="text-lg font-extrabold">{draft.products.some((p) => p.id === edit.id) ? "Editar produto" : "Novo produto"}</h3><button onClick={() => setEdit(null)} aria-label="Fechar"><X className="h-5 w-5" /></button></div>
        {edit.image && <img src={edit.image} alt="" className="h-40 w-full rounded-2xl object-cover" />}
        {[["name", "Nome"], ["image", "Link da foto"], ["description", "Descrição"]].map(([k, l]) => <label key={k} className="block text-xs font-semibold text-gray-500">{l}<input value={edit[k] || ""} onChange={(e) => setEdit(Object.assign({}, edit, { [k]: e.target.value }))} className={inp} /></label>)}
        <div className="grid grid-cols-2 gap-3">{[["price", "Preço (R$)"], ["oldPrice", "Preço antigo (R$)"], ["stock", "Estoque"], ["rating", "Nota (0-5)"]].map(([k, l]) => <label key={k} className="block text-xs font-semibold text-gray-500">{l}<input type="number" step="0.01" value={edit[k]} onChange={(e) => setEdit(Object.assign({}, edit, { [k]: Number(e.target.value) }))} className={inp} /></label>)}</div>
        <label className="block text-xs font-semibold text-gray-500">Categoria<select value={edit.category} onChange={(e) => setEdit(Object.assign({}, edit, { category: e.target.value }))} className={inp}>{draft.categories.map((k) => <option key={k.name}>{k.name}</option>)}</select></label>
        <label className="flex items-center gap-2 text-sm font-semibold"><input type="checkbox" checked={edit.active !== false} onChange={(e) => setEdit(Object.assign({}, edit, { active: e.target.checked }))} /> Visível na loja</label>
        <div className="flex gap-2">{draft.products.some((p) => p.id === edit.id) && <button onClick={() => { const next = Object.assign({}, draft, { products: draft.products.filter((p) => p.id !== edit.id) }); setEdit(null); save(next); }} className="h-12 rounded-xl border px-4 text-sm font-bold text-red-600">Excluir</button>}
          <button disabled={saving || !edit.name.trim()} onClick={() => { const exists = draft.products.some((p) => p.id === edit.id); const next = Object.assign({}, draft, { products: exists ? draft.products.map((p) => p.id === edit.id ? edit : p) : [edit].concat(draft.products) }); setEdit(null); save(next); }} className="h-12 flex-1 rounded-xl bg-gray-900 font-bold text-white">{saving ? "Salvando…" : "Salvar produto"}</button></div>
      </div></div>}
      {msg && <div className="fixed bottom-6 left-1/2 z-[60] -translate-x-1/2 rounded-full bg-gray-900 px-5 py-3 text-sm font-bold text-white shadow-2xl">{msg}</div>}
    </div>
  );
}

function App() {
  const cat = useCatalog();
  const [hash, setHash] = useState(typeof location !== "undefined" ? location.hash : "");
  useEffect(() => {
    const l = document.createElement("link"); l.rel = "stylesheet"; l.href = "https://fonts.googleapis.com/css2?family=Plus+Jakarta+Sans:wght@400;600;700;800&display=swap"; document.head.appendChild(l);
    const on = () => setHash(location.hash); window.addEventListener("hashchange", on); return () => window.removeEventListener("hashchange", on);
  }, []);
  if (!cat.loaded) return <div className="flex min-h-screen items-center justify-center text-sm text-gray-500">Carregando loja…</div>;
  return SITE_MODE === "admin" || hash === "#/admin" ? <Admin cat={cat} /> : <Store cat={cat} />;
}
